Модель и хранилище

Models/Cargo.cs

public class Cargo
{
    public int Id { get; set; }

    [Required, StringLength(100)]
    public string Name { get; set; } = string.Empty;

    [Required, StringLength(60)]
    public string Category { get; set; } = string.Empty;

    [Range(1, 1_000_000)]
    public int Quantity { get; set; } = 1;

    [Required, StringLength(100)]
    public string Location { get; set; } = string.Empty;

    [Range(0.01, 1_000_000_000)]
    public decimal Weight { get; set; }

    [Required, StringLength(300)]
    public string Description { get; set; } = string.Empty;

    public bool IsAvailable { get; set; } = true;

    public decimal TotalWeight => Math.Round(Weight * Quantity, 2);

    public string Status => IsAvailable ? "На складе" : "Нет в наличии";
}

Data/CargoStore.cs — фрагмент

public static class CargoStore
{
    public static object SyncRoot { get; } = new();

    public static List<Cargo> Cargoes { get; } = new()
    {
        new Cargo
        {
            Id = 1,
            Name = "Ноутбуки Dell Latitude 5440",
            Category = "Электроника",
            Quantity = 24,
            Location = "Сектор A-1, стеллаж 3",
            Weight = 2.4m,
            Description = "Партия офисных ноутбуков...",
            IsAvailable = true
        },
        // ... всего шесть начальных объектов
    };

    public static int GetNextId() =>
        Cargoes.Count == 0 ? 1 : Cargoes.Max(cargo => cargo.Id) + 1;
}
  • Одна коллекция используется всеми endpoint'ами, поэтому изменения сразу видны при следующем запросе.
  • GetNextId() назначает идентификатор на сервере: Max(Id) + 1.
  • SyncRoot защищает коллекцию от одновременных изменений из параллельных HTTP-запросов.
  • Вычисляемые TotalWeight и Status попадают в JSON-ответ, но не принимаются из тела запроса.

Query-параметр search ищет по названию, категории, месту хранения и описанию без учёта регистра. Дополнительно поддерживаются category и available.

GET /api/cargoes — список и фильтрация

app.MapGet("/api/cargoes", (string? search, string? category, bool? available) =>
{
    List<Cargo> snapshot;
    lock (CargoStore.SyncRoot)
    {
        snapshot = CargoStore.Cargoes.ToList();
    }

    IEnumerable<Cargo> cargoes = snapshot;

    if (!string.IsNullOrWhiteSpace(search))
    {
        var query = search.Trim();
        cargoes = cargoes.Where(cargo =>
            cargo.Name.Contains(query, StringComparison.OrdinalIgnoreCase) ||
            cargo.Category.Contains(query, StringComparison.OrdinalIgnoreCase) ||
            cargo.Location.Contains(query, StringComparison.OrdinalIgnoreCase) ||
            cargo.Description.Contains(query, StringComparison.OrdinalIgnoreCase));
    }

    if (available is not null)
    {
        cargoes = cargoes.Where(cargo => cargo.IsAvailable == available);
    }

    return Results.Ok(cargoes.OrderBy(cargo => cargo.Id));
});

GET по id и POST

GET /api/cargoes/{id}

app.MapGet("/api/cargoes/{id:int}", (int id) =>
{
    Cargo? cargo;
    lock (CargoStore.SyncRoot)
    {
        cargo = CargoStore.Cargoes.FirstOrDefault(item => item.Id == id);
    }

    return cargo is null
        ? Results.NotFound()
        : Results.Ok(cargo);
});

Если объект не найден, возвращается 404 Not Found; иначе — 200 OK с JSON-объектом груза.

POST /api/cargoes — создание

app.MapPost("/api/cargoes", (Cargo input) =>
{
    Normalize(input);

    var errors = Validate(input);
    if (errors.Count > 0)
    {
        return Results.ValidationProblem(errors);
    }

    lock (CargoStore.SyncRoot)
    {
        input.Id = CargoStore.GetNextId();
        CargoStore.Cargoes.Add(input);
    }

    return Results.Created($"/api/cargoes/{input.Id}", input);
});

Перед сохранением выполняется Normalize(input) — обрезка пробелов — и валидация по атрибутам DataAnnotations. Идентификатор назначается на сервере, а заголовок Location указывает на созданный ресурс.

PUT и DELETE

PUT /api/cargoes/{id} — изменение

app.MapPut("/api/cargoes/{id:int}", (int id, Cargo input) =>
{
    Normalize(input);

    lock (CargoStore.SyncRoot)
    {
        var cargo = CargoStore.Cargoes.FirstOrDefault(item => item.Id == id);
        if (cargo is null)
        {
            return Results.NotFound();
        }

        var errors = Validate(input);
        if (errors.Count > 0)
        {
            return Results.ValidationProblem(errors);
        }

        cargo.Name = input.Name;
        cargo.Category = input.Category;
        cargo.Quantity = input.Quantity;
        cargo.Location = input.Location;
        cargo.Weight = input.Weight;
        cargo.Description = input.Description;
        cargo.IsAvailable = input.IsAvailable;

        return Results.Ok(cargo);
    }
});

DELETE /api/cargoes/{id} — удаление

app.MapDelete("/api/cargoes/{id:int}", (int id) =>
{
    lock (CargoStore.SyncRoot)
    {
        var cargo = CargoStore.Cargoes.FirstOrDefault(item => item.Id == id);
        if (cargo is null)
        {
            return Results.NotFound();
        }

        CargoStore.Cargoes.Remove(cargo);
    }

    return Results.NoContent();
});

PUT заменяет все изменяемые поля объекта и возвращает обновлённый груз с пересчитанными TotalWeight и Status. DELETE удаляет объект из коллекции и отвечает 204 No Content без тела ответа.

OpenAPI и Swagger UI

OpenAPI — это спецификация API в формате JSON. В .NET 9 документ генерируется автоматически по endpoint'ам и их метаданным. Swagger UI — графическая оболочка, которая читает документ и позволяет выполнять запросы. 

builder.Services.AddOpenApi(options =>
{
    options.AddDocumentTransformer((document, _, _) =>
    {
        document.Info = new OpenApiInfo
        {
            Title = "CargoFlow API",
            Version = "v1",
            Description = "REST API для учёта грузов на складе."
        };

        // Относительный адрес сервера: Swagger UI обращается к тому же
        // протоколу и хосту, на котором открыта документация.
        document.Servers = new List<OpenApiServer>
        {
            new() { Url = "/" }
        };

        return Task.CompletedTask;
    });
});

app.MapOpenApi();                       // /openapi/v1.json

app.UseSwaggerUI(options =>
{
    options.RoutePrefix = "swagger";    // /swagger
    options.SwaggerEndpoint("/openapi/v1.json", "CargoFlow API v1");
});

Документ содержит пути, параметры маршрута и query-параметры, тела запросов, ответы 200/201/204/400/404 и схемы Cargo и HttpValidationProblemDetails. Метаданные задаются методами WithSummary, WithDescription, Accepts, Produces и ProducesValidationProblem. Дополнительно подключён Scalar UI: /scalar/v1.

Docker

Используется многоэтапная сборка: SDK-образ нужен только для restore и publish, а в итоговый runtime-образ попадают лишь опубликованные файлы. Приложение слушает 0.0.0.0 и читает порт из переменной окружения PORT. 

FROM mcr.microsoft.com/dotnet/sdk:9.0 AS build
WORKDIR /src

COPY wt-lab6-6minimalapi-yushkevich.csproj ./
RUN dotnet restore

COPY . .
RUN dotnet publish wt-lab6-6minimalapi-yushkevich.csproj \
    -c Release \
    -o /app/publish \
    /p:UseAppHost=false

FROM mcr.microsoft.com/dotnet/aspnet:9.0 AS final
WORKDIR /app

COPY --from=build /app/publish .

ENV ASPNETCORE_URLS=http://0.0.0.0:10000
ENV PORT=10000
EXPOSE 10000

ENTRYPOINT ["dotnet", "wt-lab6-6minimalapi-yushkevich.dll"]

Заключение

В ходе лабораторной работы разработан и опубликован REST API для учёта грузов на складе. Приложение реализовано на ASP.NET Core Minimal API, работает с JSON, поддерживает полный набор CRUD-операций и использует корректные HTTP-статусы.

Что освоено

  • Создание проекта Minimal API и регистрация endpoint'ов через MapGet, MapPost, MapPut, MapDelete.
  • Привязка параметров маршрута, query-параметров и тела запроса JSON.
  • Возврат результатов через Results.Ok, Results.Created, Results.NoContent, Results.NotFound и Results.ValidationProblem.
  • Валидация модели по атрибутам DataAnnotations и формирование 400 Bad Request.
  • Автоматическая генерация OpenAPI-документа, подключение Swagger UI и Scalar UI.
  • Многоэтапная Docker-сборка, запуск контейнера и публикация на Render.
  • Работа с Git: осмысленные коммиты, удалённый репозиторий, повторный deployment.

Ограничения

Данные хранятся в памяти процесса, поэтому после перезапуска приложения и нового deployment на Render коллекция восстанавливается из начального набора. Несколько экземпляров приложения не разделяют данные между собой. Для перехода к постоянному хранению потребуется заменить CargoStore на слой доступа к PostgreSQL (например, Entity Framework Core), добавить строку подключения и миграции.
